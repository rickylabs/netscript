async function copyTree(source: string, destination: string) {
  await Deno.mkdir(root + '/' + destination, {recursive: true});
  for await (const entry of Deno.readDir(source)) {
    if (entry.isSymlink) throw new Error('Unexpected evidence symlink: ' + entry.name);
    const from = source + '/' + entry.name;
    const to = destination + '/' + entry.name + (/\.tsx?$/.test(entry.name) ? '.txt' : '');
    if (entry.isDirectory) { await copyTree(from, destination + '/' + entry.name); continue; }
    const bytes = await Deno.readFile(from);
    await Deno.writeFile(root + '/' + to, bytes, {createNew: true});
    const actual = await sha(await Deno.readFile(root + '/' + to));
    if (actual !== await sha(bytes)) throw new Error('Archive byte mismatch: ' + to);
    rows.push({source: from, archive: to, bytes: bytes.length, sha256: actual});
    scoped.push(to);
  }
}
