const bindings: Record<string, unknown>[] = [];
async function bind(path: string, content: string, producer: string, source?: string): Promise<void> {
  const actual = await Deno.readTextFile(join(candidate, path));
  const sourceMatches = source ? (await Deno.readTextFile(source)).replaceAll('\r\n', '\n') === content : undefined;
  bindings.push({ path, producer, matches: actual === content, ...(source ? { source, sourceMatches } : {}) });
  if (actual !== content || sourceMatches === false) throw new Error(`Producer binding failed: ${path}`);
}
