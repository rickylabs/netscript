function require(value: unknown, message: string): asserts value {

}
async function sha(path: string): Promise<string> {
  return new Uint8Array(await crypto.subtle.digest('SHA-256', await Deno.readFile(path))).toHex();
}
async function tree(root: string): Promise<Record<string, string>> {
  const files: Record<string, string> = {};
  async function visit(current: string): Promise<void> {
    for await (const entry of Deno.readDir(current)) {
      require(!entry.isSymlink, 'Native project contains an unexpected symlink');
      const path = join(current, entry.name);
      if (entry.isDirectory) await visit(path);
      else if (entry.isFile) files[relative(root, path).replaceAll('\\', '/')] = await sha(path);
    }
  }
  await visit(root);
  return Object.fromEntries(Object.entries(files).sort(([a], [b]) => a.localeCompare(b)));
}
Deno.Command = class extends OriginalCommand {
  readonly command: string | URL;
  readonly options?: Deno.CommandOptions;
  constructor(command: string | URL, options?: Deno.CommandOptions) {
    super(command, options);
    this.command = command;
    this.options = options;
  }
  override async output(): Promise<Deno.CommandOutput> {
    const startedAt = new Date().toISOString();
    try {
      const result = await super.output();
      commands.push({ command: String(this.command), args: this.options?.args, startedAt,
        finishedAt: new Date().toISOString(), code: result.code, signal: result.signal,
        stdout: decoder.decode(result.stdout), stderr: decoder.decode(result.stderr) });
      return result;
    } catch (error) {
      commands.push({ command: String(this.command), args: this.options?.args, startedAt,
        finishedAt: new Date().toISOString(), error: error instanceof Error ? error.message : String(error) });
      throw error;
    }
  }
};
