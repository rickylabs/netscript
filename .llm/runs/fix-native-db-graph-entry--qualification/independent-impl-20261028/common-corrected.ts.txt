export function decode(bytes: Uint8Array): string { return new TextDecoder('utf-8', { fatal: true }).decode(bytes); }
export async function hash(path: string): Promise<string> { return digest(await Deno.readFile(path)); }
export async function digest(bytes: Uint8Array): Promise<string> {
  return [...new Uint8Array(await crypto.subtle.digest('SHA-256', new Uint8Array(bytes)))]
    .map(n => n.toString(16).padStart(2, '0')).join('');
}

export async function git(args: string[], cwd: string = root): Promise<string> {
  const out = await new Deno.Command('git', { args, cwd, stdout: 'piped', stderr: 'null' }).output();
  if (!out.success) throw new Error(`Git guard failed: ${args[0]}`);
  return decode(out.stdout).trim();
}
export async function guard(): Promise<void> {
  const snapshot = JSON.parse(await Deno.readTextFile(`${directory}/source-snapshot.json`));
  for(const target of snapshot.targets){
    if(await git(['rev-parse','HEAD'],target.root)!==target.head)throw new Error(`Exact ${target.name} head changed.`);
    await git(['diff','--quiet','HEAD','--'],target.root);
    for(const item of target.protected){if(await hash(`${target.root}/${item.path}`)!==item.sha256)throw new Error(`Protected ${target.name} drift: ${item.path}`);}
    const extra=(await git(['ls-files','--others','--exclude-standard'],target.root)).split(/\r?\n/).filter(Boolean);
    for(const path of extra){if(!target.untracked.includes(path)){
      if(path!=='.netscript/agent/diagnostics/project.json')throw new Error(`Unexpected ${target.name} untracked owner.`);
      const receipt=JSON.parse(await Deno.readTextFile(`${target.root}/${path}`));
      if(Object.keys(receipt).sort().join(',')!=='command,exitStatus,resource,timestamp'||receipt.resource!=='project'||

}

export async function finiteJson<T>(client: Deno.HttpClient, path: string, key?: string, limit = 262144): Promise<T> {
  const init: RequestInit & {client:Deno.HttpClient} = {client,redirect:'error',
    signal:AbortSignal.timeout(15000),headers:key ? {authorization:`Bearer ${key}`} : {}};
  const response = await fetch(`https://openrouter.ai/api/v1/${path}`, init);
  if (!response.ok || !response.body) { await response.body?.cancel(); throw new Error(`Provider metadata HTTP${response.status}`); }
  const reader = response.body.getReader();
  const chunks: Uint8Array[] = []; let size = 0;
  try {
    while (true) {
      const next = await reader.read(); if (next.done) break;
      size += next.value.length;
      if (size > limit) { await reader.cancel(); throw new Error('Provider metadata streaming bound exceeded.'); }
      chunks.push(next.value);
    }
  } finally { reader.releaseLock(); }
  const bytes = new Uint8Array(size); let offset = 0;
  for (const chunk of chunks) { bytes.set(chunk, offset); offset += chunk.length; }
  return JSON.parse(decode(bytes));
}
export interface Allowance {
  provider:'openrouter'; capturedAt:string; availableBalanceUsd:number; accountAvailableUsd:number;
  keyLimit:number|null; keyRemainingUsd:number; minimumAccountAndSelectedKey:boolean;
}
export async function allowance(client: Deno.HttpClient, key: string): Promise<Allowance> {
  const [credits, metadata] = await Promise.all([
    finiteJson<{data:{total_credits:number;total_usage:number}}>(client, 'credits', key, 65536),
    finiteJson<{data:{limit:number|null;limit_remaining:number}}>(client, 'key', key, 65536)]);
  const account = Number(credits.data.total_credits) - Number(credits.data.total_usage);
  let keyRemaining = account;
  if (metadata.data.limit !== null) {
    keyRemaining = metadata.data.limit_remaining;
    if (typeof keyRemaining !== 'number' || !Number.isFinite(keyRemaining) || keyRemaining < 0) throw new Error('Key cap unproven.');
  }
  const available = Math.min(account, keyRemaining);
  if (!Number.isFinite(available) || available < 0) throw new Error('Account/key allowance unproven.');
  return { provider:'openrouter' as const, capturedAt:new Date().toISOString(), availableBalanceUsd:available,
    accountAvailableUsd:account, keyLimit:metadata.data.limit, keyRemainingUsd:keyRemaining,
    minimumAccountAndSelectedKey:true };
}
export interface Price {
  capturedAt:string; inputMaxUsdPerMillion:number;outputMaxUsdPerMillion:number;steps:number;
  inputPerStep:number;outputPerStep:number;prospectiveUsd:number;estimateUsd:number;noCacheDiscount:boolean;
  noProviderPin:boolean;estimateNotProviderCap:boolean;nativeCatalogLimitNotEffectiveOutputCap:boolean;
}
export async function price(client: Deno.HttpClient, name: string): Promise<Price> {
  const response = await finiteJson<{data:{endpoints:{pricing:{prompt:string;completion:string}}[]}}>(client, 'models/z-ai/glm-5.3-flash/endpoints');
  await json(name, response);
  const endpoints = response.data?.endpoints;
  if (!Array.isArray(endpoints) || !endpoints.length) throw new Error('Concrete endpoint catalog empty.');
  const input = Math.max(...endpoints.map(entry => Number(entry.pricing.prompt) * 1000000));
  const output = Math.max(...endpoints.map(entry => Number(entry.pricing.completion) * 1000000));
  const prospective = ((steps * 96000 * input + steps * 32000 * output) / 1000000 + 0.10) * 1.25;
  if (![input, output, prospective].every(n => Number.isFinite(n) && n > 0)) throw new Error('Price/positive estimate unproven.');
  return { capturedAt:new Date().toISOString(), inputMaxUsdPerMillion:input, outputMaxUsdPerMillion:output,
    steps, inputPerStep:96000, outputPerStep:32000, prospectiveUsd:prospective,
    estimateUsd:Math.ceil(prospective * 10) / 10, noCacheDiscount:true, noProviderPin:true,
    estimateNotProviderCap:true, nativeCatalogLimitNotEffectiveOutputCap:true };
}
